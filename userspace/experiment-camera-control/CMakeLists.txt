cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(experiment_camera_control_tests LANGUAGES CXX)
    enable_testing()
    foreach(suite shell camera exposure runtime rx)
        add_executable(${suite}_test tests/${suite}_test.cpp)
        target_include_directories(${suite}_test PRIVATE src)
        target_compile_features(${suite}_test PRIVATE cxx_std_17)
        target_compile_options(${suite}_test PRIVATE -Wall -Wextra -Werror)
        add_test(NAME ${suite} COMMAND ${suite}_test)
    endforeach()
    add_executable(isp_test tests/isp_test.cpp src/isp_camera.cpp)
    target_include_directories(isp_test PRIVATE src tests/mock)
    target_compile_features(isp_test PRIVATE cxx_std_17)
    target_compile_options(isp_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME isp_backend COMMAND isp_test)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME link_audit COMMAND ${Python3_EXECUTABLE} -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests")
else()
    include("${CMAKE_CURRENT_SOURCE_DIR}/firmware.cmake")
endif()
