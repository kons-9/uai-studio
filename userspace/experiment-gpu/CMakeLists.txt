cmake_minimum_required(VERSION 3.16)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	set(EXPERIMENT_EXTENSION "${CMAKE_CURRENT_SOURCE_DIR}/src/firmware.cpp")
	include("${CMAKE_CURRENT_LIST_DIR}/firmware.cmake")
	return()
endif()
project(experiment_gpu_tests LANGUAGES CXX)
enable_testing()
add_executable(blit_test tests/blit_test.cpp)
target_include_directories(blit_test PRIVATE src)
target_compile_features(blit_test PRIVATE cxx_std_17)
target_compile_options(blit_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME blit COMMAND blit_test)
add_executable(dma2d_test tests/dma2d_test.cpp)
target_include_directories(dma2d_test PRIVATE tests/mock src)
target_compile_features(dma2d_test PRIVATE cxx_std_17)
target_compile_options(dma2d_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME dma2d COMMAND dma2d_test)
add_executable(scenario_test tests/scenario_test.cpp src/firmware.cpp)
target_sources(scenario_test PRIVATE src/camera_runtime/middleware/ui/canvas.cpp)
target_include_directories(scenario_test PRIVATE
	tests/mock src src/camera_runtime)
target_compile_features(scenario_test PRIVATE cxx_std_17)
target_compile_options(scenario_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME scenario COMMAND scenario_test)
