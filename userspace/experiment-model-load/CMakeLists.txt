cmake_minimum_required(VERSION 3.16)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	if(NOT EXPERIMENT_MODEL_MANIFEST OR NOT EXISTS "${EXPERIMENT_MODEL_MANIFEST}")
		message(FATAL_ERROR "Set EXPERIMENT_MODEL_MANIFEST to a packed manifest matching the reserved PSRAM slots")
	endif()
	set(EXPERIMENT_EXTENSION "${CMAKE_CURRENT_SOURCE_DIR}/src/firmware.cpp")
	include("${CMAKE_CURRENT_LIST_DIR}/firmware.cmake")
	execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expected.py"
		"${EXPERIMENT_MODEL_MANIFEST}" "${CMAKE_CURRENT_BINARY_DIR}/model_expected.hpp"
		RESULT_VARIABLE expected_result ERROR_VARIABLE expected_error)
	if(NOT expected_result EQUAL 0)
		message(FATAL_ERROR "Invalid model expectation: ${expected_error}")
	endif()
	target_include_directories(${TARGET_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
	target_link_options(${TARGET_NAME} BEFORE PRIVATE "-T${CMAKE_CURRENT_SOURCE_DIR}/slots.ld")
	set_property(TARGET ${TARGET_NAME} APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/slots.ld")
	return()
endif()
project(experiment_model_load_tests LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(manifest_probe tests/manifest_probe.cpp)
target_include_directories(manifest_probe PRIVATE src)
target_compile_features(manifest_probe PRIVATE cxx_std_17)
target_compile_options(manifest_probe PRIVATE -Wall -Wextra -Werror)
add_test(NAME manifest COMMAND ${Python3_EXECUTABLE} -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests")
add_executable(staging_test tests/staging_test.cpp)
target_include_directories(staging_test PRIVATE src)
target_compile_features(staging_test PRIVATE cxx_std_17)
target_compile_options(staging_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME staging COMMAND staging_test)
add_executable(protocol_test tests/protocol_test.cpp)
target_include_directories(protocol_test PRIVATE src src/camera_runtime)
target_compile_features(protocol_test PRIVATE cxx_std_17)
target_compile_options(protocol_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME protocol COMMAND protocol_test)
set_tests_properties(manifest PROPERTIES ENVIRONMENT "MANIFEST_PROBE=$<TARGET_FILE:manifest_probe>")
